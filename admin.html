전용 스키마 대신 public + book_records 를 사용합니다

최신 형제 서비스(bible 등)와 동일하게 이 프로젝트는 public 스키마에 서비스 접두사 테이블(public.book_records)을 씁니다. 공통 로그인 모듈(cg-auth.js)의 기록 헬퍼가 public 을 기본으로 하므로 전용 스키마를 만들지 않아 별도 Supabase 설정이 필요 없습니다.

Supabase SQL 가이드

Supabase 대시보드 > SQL Editor 에서 _shared/sql/01~06 을 먼저 실행한 뒤 아래 08-book.sql(book/supabase/migrations/001_book_schema.sql) 을 실행하면 기록 테이블·RLS·권한이 생성됩니다.

-- 독서야 놀자 기록 테이블 (public.book_records) — 요약본. 전체는 _shared/sql/08-book.sql
create table if not exists public.book_records (
  id             uuid primary key default gen_random_uuid(),
  user_id        uuid not null references auth.users(id) on delete cascade,
  title          text not null,
  author         text, publisher text, isbn text, category text,
  reading_status text not null default 'done',
  started_at date, finished_at date, reading_date date,
  rating int,
  one_line_review text, summary text, memorable_sentence text,
  feeling text, learned text, action_plan text,
  favorite_character text, question_after_reading text,
  visibility text not null default 'private',        -- private | public
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz                               -- soft delete
);
alter table public.book_records enable row level security;

-- 공개 기록은 누구나 / 본인 기록은 전권
create policy cg_book_records_public on public.book_records
  for select to anon, authenticated
  using (visibility = 'public' and deleted_at is null);
create policy cg_book_records_own on public.book_records
  for all to authenticated
  using (auth.uid() = user_id) with check (auth.uid() = user_id);
-- 관리자 조회/운영(비공개 전환)
create policy cg_book_records_select_admin on public.book_records
  for select using (public.cg_is_admin());
create policy cg_book_records_update_admin on public.book_records
  for update using (public.cg_is_admin()) with check (public.cg_is_admin());

-- updated_at 자동 갱신 (06 에서 만든 공용 함수)
create trigger book_records_touch before update on public.book_records
  for each row execute function public.cg_touch_updated_at();

grant select on public.book_records to anon;
grant select, insert, update, delete on public.book_records to authenticated;
grant all on public.book_records to service_role;
